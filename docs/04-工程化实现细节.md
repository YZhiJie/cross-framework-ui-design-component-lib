## 二、双框架并行构建配置

### 2.1 构建策略

每个包需输出三种产物：

| 产物 | Format | 用途 | 文件后缀 |
|------|--------|------|---------|
| ESM | ES Module | 现代打包器（Vite/Webpack5+/Rollup） | `.mjs` / `.js` |
| CJS | CommonJS | Node.js require / 旧打包器 | `.cjs` |
| UMD | UMD | `<script>` 标签直接引入 | `.umd.js` |
| Types | `.d.ts` | TypeScript 类型提示 | `.d.ts` |

### 2.2 @xui-design/shared 构建配置

```typescript
// packages/shared/build.config.ts
import { defineConfig } from 'vite';
import dts from 'vite-plugin-dts';
import { resolve } from 'path';

export default defineConfig({
  plugins: [
    dts({
      entryRoot: 'src',
      outDir: 'dist/types',
      // 为每个子模块单独生成类型声明
      include: ['src/**/*.ts'],
      exclude: ['src/**/__tests__/**'],
    }),
  ],
  build: {
    lib: {
      entry: {
        // 按组件分包入口，支持按需加载
        index: resolve(__dirname, 'src/index.ts'),
        'components/button': resolve(__dirname, 'src/components/button/index.ts'),
        'components/input': resolve(__dirname, 'src/components/input/index.ts'),
        'components/modal': resolve(__dirname, 'src/components/modal/index.ts'),
        'hooks': resolve(__dirname, 'src/hooks/index.ts'),
        'utils': resolve(__dirname, 'src/utils/index.ts'),
      },
      formats: ['es', 'cjs'],
      fileName: (format, entryName) =>
        `${entryName}.${format === 'es' ? 'mjs' : 'cjs'}`,
    },
    rollupOptions: {
      // 不打包外部依赖
      external: [],
      output: {
        // 保留模块结构，支持 Tree Shaking
        preserveModules: true,
        preserveModulesRoot: 'src',
        dir: 'dist',
      },
    },
  },
});
```

### 3.3 双端类型对齐校验

```typescript
// packages/utils/src/type-check.ts
// CI 中运行的类型对齐校验脚本

import { parse } from 'ts-morph';

interface SpecContract {
  componentName: string;
  props: Record<string, string>;
  events: string[];
  slots: string[];
}

/**
 * 从 @xui-design/core 的 ComponentSpec 提取契约
 * 与 @xui-design/vue 和 @xui-design/react 的实际实现对比
 */
async function checkContractAlignment() {
  const coreSpec = await extractSpecFromCore('Button');
  const vueProps = await extractPropsFromVue('Button');
  const reactProps = await extractPropsFromReact('Button');

  const errors: string[] = [];

  // 校验 Props 一致性
  for (const [key, type] of Object.entries(coreSpec.props)) {
    if (!vueProps[key]) errors.push(`Vue: 缺少 prop "${key}"`);
    if (!reactProps[key]) errors.push(`React: 缺少 prop "${key}"`);
    if (vueProps[key] && vueProps[key] !== type) {
      errors.push(`Vue: prop "${key}" 类型不一致 (期望 ${type}, 实际 ${vueProps[key]})`);
    }
    if (reactProps[key] && reactProps[key] !== type) {
      errors.push(`React: prop "${key}" 类型不一致 (期望 ${type}, 实际 ${reactProps[key]})`);
    }
  }

  if (errors.length > 0) {
    console.error('❌ 契约对齐失败:');
    errors.forEach(e => console.error(`  - ${e}`));
    process.exit(1);
  } else {
    console.log('✅ 契约对齐通过');
  }
}

checkContractAlignment();
```

## 四、按需加载支持

### 4.1 多入口按需加载

通过 `exports` 字段为每个组件提供独立入口，消费方按需 import：

```typescript
// 方式一：从子路径引入（推荐）
import { Button } from '@xui-design/vue/components/button';
import { Input } from '@xui-design/react/components/input';

// 方式二：从主入口引入（配合 Tree Shaking）
import { Button, Input } from '@xui-design/vue';
```

两种方式都支持，但方式一显式指定子路径，打包器无需分析整棵依赖树，构建更快。

### 4.2 自动按需加载插件

#### Vue 端 — unplugin-vue-components

```typescript
// vite.config.ts (消费方项目)
import Components from 'unplugin-vue-components/vite';
import { XuiResolver } from '@xui-design/vue/resolver';

export default defineConfig({
  plugins: [
    Components({
      resolvers: [XuiResolver()],
    }),
  ],
});
```

```typescript
// @xui-design/vue/src/resolver.ts
import type { ComponentResolver } from 'unplugin-vue-components/types';

export function XuiResolver(): ComponentResolver {
  return {
    type: 'component',
    resolve: (name: string) => {
      if (name.match(/^X[A-Z]/)) {
        const componentName = name.slice(1); // 'XButton' → 'Button'
        return {
          name: componentName,
          from: '@xui-design/vue',
          sideEffects: `@xui-design/styles/components/${componentName.toLowerCase()}.module.css`,
        };
      }
    },
  };
}
```

#### React 端 — babel-plugin-import 或手动子路径

```json
// .babelrc (消费方项目)
{
  "plugins": [
    ["babel-plugin-import", {
      "libraryName": "@xui-design/react",
      "libraryDirectory": "components",
      "camel2DashComponentName": false
    }]
  ]
}

// 转换前
import { Button } from '@xui-design/react';
// 转换后
import Button from '@xui-design/react/components/button';
```

### 4.3 样式按需加载

```scss
// 每个组件有独立 CSS Module 文件
// 消费方引入组件时，打包器自动引入对应样式
```

## 五、Tree Shaking 优化策略

### 5.1 Tree Shaking 工作原理

```
消费者: import { Button } from '@xui-design/vue'

打包器分析:
  1. 读取 @xui-design/vue/package.json 的 exports + sideEffects
  2. 如果 sideEffects: false → 所有未使用的导出会被 shake 掉
  3. 如果有 sideEffects: ["*.css"] → CSS 文件保留，JS 按需引入
  4. preserveModules: true → 保留原始模块结构，精确 shake
```

### 5.2 Tree Shaking 友好的代码编写规范

#### ✅ 正确：纯导出 + 副作用隔离

```typescript
// button/index.ts
export { default as Button } from './button.vue';
export type { ButtonProps } from '@xui-design/core/components/button';

// 无任何顶层副作用代码（如 console.log、全局变量赋值）
```

#### ❌ 错误：顶层副作用

```typescript
// ❌ 这会破坏 Tree Shaking

// ❌ 修改全局对象

```

#### ✅ 正确：副作用集中在函数体内

```typescript
// 副作用只在调用时执行
export function createButton() {
  // 这里的副作用不会影响 Tree Shaking
  console.log('Button created');
  return Button;
}
```

### 5.3 sideEffects 配置策略

```json
// 三个包的 sideEffects 配置

    "*.vue",

    "*.scss"
  ]

  "sideEffects": [
    "*.css",
    "*.scss"
  ]

// @xui-design/styles — 全部是样式，有副作用
{
  "sideEffects": [
    "*.css",
    "*.scss",
    "*.sass"
  ]
}
```

### 5.4 产物体积检查

构建后自动检查每个组件的产物体积：

```typescript
// scripts/check-bundle-size.ts
import { statSync } from 'fs';
import { gzipSync } from 'zlib';
import { readFileSync } from 'fs';

interface SizeLimit {
  path: string;
  limit: number; // KB
}

const limits: SizeLimit[] = [
  { path: 'packages/vue/dist/components/button/index.mjs', limit: 3 },
  { path: 'packages/vue/dist/components/input/index.mjs', limit: 3 },
  { path: 'packages/vue/dist/components/modal/index.mjs', limit: 5 },
  { path: 'packages/react/dist/components/button/index.mjs', limit: 3 },
  { path: 'packages/shared/dist/index.mjs', limit: 10 },
];

let passed = true;
for (const { path, limit } of limits) {
  const content = readFileSync(path);
  const gzipped = gzipSync(content);
  const sizeKB = gzipped.length / 1024;
  const status = sizeKB <= limit ? '✅' : '❌';
  if (sizeKB > limit) passed = false;
  console.log(`${status} ${path}: ${sizeKB.toFixed(2)}KB (limit: ${limit}KB)`);
}

if (!passed) process.exit(1);
```
